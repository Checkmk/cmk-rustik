13#[derive(Debug, Serialize)]
14#[serde(untagged)]
15pub(crate) enum KubeletHealth {
16    Response { status_code: u16, response: String },
17    ConnectionError { message: String },
18}
19
20pub(crate) struct KubeletHealthScraper {
21    scrape_client: Client,
22    relay_client: Client,
23    args: Arc<CliArgs>,
24}
25
26impl KubeletHealthScraper {
27    pub(crate) fn new(args: Arc<CliArgs>, metrics_cache_client: Client) -> KubeletHealthScraper {
28        let scrape_client = Client::builder()
29            .danger_accept_invalid_certs(true)
30            .build()
31            .expect("Could not build scrape client for kubelet health");
32        KubeletHealthScraper {
33            scrape_client,
34            relay_client: metrics_cache_client,
35            args,
36        }
37    }
38}

40impl Scraper for KubeletHealthScraper {
41    fn relay_client(&self) -> Client {
42        self.relay_client.clone()
43    }
44
45    fn args(&self) -> Arc<CliArgs> {
46        self.args.clone()
47    }
48
49    async fn scrape(&self) -> Result<Payload> {
50        let node_ip = std::env::var("NODE_IP").map_err(|e| Error::EnvVar {
51            name: "NODE_IP".to_string(),
52            source: e,
53        })?;
54        let node_name = std::env::var("NODE_NAME").map_err(|e| Error::EnvVar {
55            name: "NODE_NAME".to_string(),
56            source: e,
57        })?;
58        let token =
59            tokio::fs::read_to_string("/var/run/secrets/kubernetes.io/serviceaccount/token")
60                .await?;
61
62        let url = kubelet::url(&node_ip, "/healthz")?;
63        let response = self
64            .scrape_client
65            .get(&url)
66            .bearer_auth(token.trim())
67            .send()
68            .await;
69
70        let health = match response {
71            Ok(response) => {
72                let status_code = response.status();
73                if status_code == StatusCode::OK {
74                    debug!(status = %status_code, url, "kubelet healthz scrape complete");
75                } else {
76                    warn!(status = %status_code, url, "kubelet healthz scrape returned non-OK status");
77                }
78                let response_body = response.text().await?;
79                KubeletHealth::Response {
80                    status_code: status_code.as_u16(),
81                    response: response_body,
82                }
83            }
84            Err(e) => KubeletHealth::ConnectionError {
85                message: e.to_string(),
86            },
87        };
88
89        Ok(Payload::KubeletHealth {
90            node_name,
91            body: Bytes::from(serde_json::to_vec(&health)?),
92        })
93    }
94}
