5pub(crate) fn url(node_ip: &str, path: &str) -> Result<String> {
6    let ip = node_ip
7        .parse::<IpAddr>()
8        .map_err(|source| Error::InvalidNodeIp {
9            value: node_ip.to_string(),
10            source,
11        })?;
12    Ok(format!("https://{}{path}", SocketAddr::new(ip, 10250)))
13}
