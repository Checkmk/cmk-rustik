15const AGENT_PATH: &str = "/usr/local/bin/check_mk_agent";

22impl LinuxAgentScraper {

30
31impl Scraper for LinuxAgentScraper {
32    fn poll_interval(&self) -> Duration {
33        self.args.system_agent_poll_interval
34    }
35
36    fn relay_client(&self) -> Client {
37        self.relay_client.clone()
38    }
39
40    fn args(&self) -> Arc<CliArgs> {
41        self.args.clone()
42    }
43
44    /// Run the local check_mk_agent script and capture its raw stdout.
45    ///
46    /// We do not parse the output or perform any calculations on it here,
47    /// leaving these tasks for metrics-cache (and later, Checkmk itself) to do.
48    async fn scrape(&self) -> Result<Payload> {
49        let node_name = std::env::var("NODE_NAME").map_err(|e| Error::EnvVar {
50            name: "NODE_NAME".to_string(),
51            source: e,
52        })?;
53
54        debug!("running check_mk_agent");
55        let child = Command::new(AGENT_PATH)
56            .env("PYTHONDONTWRITEBYTECODE", "1")
57            .stdin(Stdio::null())
58            .stdout(Stdio::piped())
59            .stderr(Stdio::piped())
60            .kill_on_drop(true)
61            .spawn()?;
62
63        let agent_timeout = self.args.system_agent_timeout;
64        let output = timeout(agent_timeout, child.wait_with_output())
65            .await
66            .map_err(|_| Error::AgentTimeout(agent_timeout))??;
67
68        if !output.status.success() {
69            return Err(Error::AgentExitStatus {
70                status: output.status,
71                stderr: String::from_utf8_lossy(&output.stderr).into_owned(),
72            });
73        }
74
75        if output.stdout.is_empty() {
76            return Err(Error::AgentEmptyOutput);
77        }
78
79        trace!(bytes = output.stdout.len(), "check_mk_agent run complete");
80        Ok(Payload::CheckmkLinuxAgent {
81            node_name,
82            body: Bytes::from(output.stdout),
83        })
84    }
85}
