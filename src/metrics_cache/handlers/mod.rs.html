11pub fn pull_app<V: TokenValidator>(state: AppState<V>, pull: PullAgentMiddlewareConfig) -> Router {
12    let routes = Router::new()
13        .route("/sections", get(pull::get))
14        .route_layer(middleware::from_fn_with_state(
15            pull,
16            auth::pull_agent::authenticate,
17        ));
18
19    Router::new()
20        .route("/", get(|| async { "foo" }))
21        .nest("/pull", routes)
22        .route("/livez", get(health::livez))
23        .route("/readyz", get(health::readyz))
24        .layer(tower_http::compression::CompressionLayer::new())
25        .with_state(state)
26}

28/// Ingestion shares state with the rest of the application, but is considered
29/// its own app for purposes of TLS termination: We want metrics-fetcher to be
30/// able to communicate to metrics-cache using a separate set of TLS credentials
31/// than anything else which talks to metrics cache (such as an Ingress).
32pub fn ingest_app<V: TokenValidator>(state: AppState<V>) -> Router {
33    let routes = Router::new()
34        .route(
35            "/kubelet_stats_summary",
36            post(ingest::kubelet_stats_summary),
37        )
38        .route("/kubelet_health/{node_name}", post(ingest::kubelet_health))
39        .route("/system_agent/{node_name}", post(ingest::system_agent))
40        .route_layer(middleware::from_fn_with_state(
41            state.clone(),
42            auth::kubernetes::authenticate,
43        ));
44
45    Router::new()
46        .nest("/ingest", routes)
47        .layer(tower_http::compression::CompressionLayer::new())
48        .with_state(state)
49}
