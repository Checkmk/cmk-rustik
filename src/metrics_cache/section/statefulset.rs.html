9#[derive(Serialize)]
10pub(crate) struct KubeStatefulSetInfoV1<'a> {
11    pub name: &'a str,
12    pub namespace: &'a str,
13    pub creation_timestamp: Option<f64>,
14    pub labels: BTreeMap<&'a str, LabelRef<'a>>,
15    pub annotations: BTreeMap<&'a str, &'a str>,
16    pub selector: Selector<'a>,
17    pub containers: ThinContainers<'a>,
18    pub cluster: &'a str,
19    pub kubernetes_cluster_hostname: &'a str,
20}
21
22impl<'a> KubeStatefulSetInfoV1<'a> {
23    pub(crate) fn from_statefulset(
24        statefulset: &'a StatefulSet,
25        containers: ThinContainers<'a>,
26        settings: &'a HostSettings,
27    ) -> Option<Self> {
28        let spec = statefulset.spec.as_ref()?;
29        Some(Self {
30            name: statefulset.metadata.name.as_deref()?,
31            namespace: statefulset.metadata.namespace.as_deref()?,
32            creation_timestamp: statefulset
33                .metadata
34                .creation_timestamp
35                .as_ref()
36                .map(|t| t.0.as_millisecond() as f64 / 1000.0),
37            labels: statefulset
38                .metadata
39                .labels
40                .as_ref()
41                .map(LabelRef::from_map)
42                .unwrap_or_default(),
43            annotations: statefulset
44                .metadata
45                .annotations
46                .as_ref()
47                .map(|annotations| settings.annotation_key_pattern.filter(annotations))
48                .unwrap_or_default(),
49            selector: Selector::from_label_selector(&spec.selector),
50            containers,
51            cluster: &settings.cluster_name,
52            kubernetes_cluster_hostname: &settings.cluster_host_name,
53        })
54    }
55}

58    const NAME: &'static str = "kube_statefulset_info_v1";
59}

61/// StatefulSet replica counts. (`kube_statefulset_replicas_v1`)
62#[derive(Serialize)]
63pub(crate) struct KubeStatefulSetReplicasV1 {
64    available: i32,
65    desired: i32,
66    ready: i32,
67    updated: i32,
68}
69
70impl KubeStatefulSetReplicasV1 {
71    pub(crate) fn from_statefulset(statefulset: &StatefulSet) -> Option<Self> {
72        let spec = statefulset.spec.as_ref()?;
73        let status = statefulset.status.as_ref()?;
74        Some(Self {
75            available: status.available_replicas.unwrap_or(0),
76            desired: spec.replicas?,
77            ready: status.ready_replicas.unwrap_or(0),
78            updated: status.updated_replicas.unwrap_or(0),
79        })
80    }
81}
