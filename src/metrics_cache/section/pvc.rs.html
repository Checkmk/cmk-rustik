57/// PVC overview. (`kube_pvc_v1`)
58///
59/// At a high level, this section represents the join of two kinds of data from
60/// the same source: The Kubernetes API.
61///
62/// A Pod requests one or more volumes (and of those volumes, we only concern
63/// ourselves with PVCs). It references these PVCs by their name (which must be
64/// Namespace-unique and the Namespace must be the same as the Pod itself).
65///
66/// Once we know the names of all the PVCs (i.e., once we have the
67/// "claim_names"), we can then look up each of the PVCs and the necessary
68/// information about it.
69///
70/// We _could_ do that lookup on the fly, but since multiple pods likely exist
71/// in the same namespace, we avoid needing to loop over the PVCs multiple times
72/// by instead caching a mapping in the [`crate::snapshot::Snapshot`]. In
73/// particular, we build a `HashMap` that is indexed by the PVC's name, which is
74/// what a pod's claim name resolves to, like this:
75/// `map[namespace][claim_name] -> PVC`.
76///
77/// Then in the implementation below, we simply take `namespace` and the list
78/// of claim names we care about (determined by [`Self::pod_pvc_claim_names()`]
79/// and [`Self::workload_pvc_claim_names()`] at the section-emit call site) and
80/// return information about each PVC keyed on its claim name.
81#[derive(Debug, Default, Serialize)]
82pub(crate) struct KubePvcV1<'a> {
83    claims: BTreeMap<String, Claim<'a>>,
84}
85
86impl<'a> KubePvcV1<'a> {
87    /// Given a snapshot, a namespace in which the claims live, and the claim
88    /// names for which we want PVC information, generate the PVC information
89    /// if able.
90    ///
91    /// In several cases, we might not "be able" to generate PVC information.
92    ///
93    /// For example, if a claim name is provided which does not exist in
94    /// `indexes` (and therefore was not known by the Kubernetes API at the
95    /// time of snapshot capture), that claim name is skipped.
96    ///
97    /// If we get to the end and would not produce information for _any_ claim
98    /// name, we return `None` so that the section-emit call site can omit the
99    /// section entirely.
100    pub fn from_claim_names(
101        indexes: &'a Indexes,
102        namespace: &'a str,
103        claim_names: impl IntoIterator<Item = &'a str>,
104    ) -> Option<KubePvcV1<'a>> {
105        let mut out = Self::default();
106        for claim_name in claim_names {
107            let Some(pvc) = indexes.pvc(namespace, claim_name) else {
108                continue;
109            };
110            let capacity = pvc
111                .status
112                .as_ref()
113                .and_then(|s| s.capacity.as_ref())
114                .and_then(|b| b.get("storage"))
115                .and_then(|q| parse_quantity(&q.0))
116                .map(|v| StorageRequirement { storage: v as u64 });
117            let claim = Claim {
118                metadata: Metadata {
119                    name: claim_name,
120                    namespace,
121                },
122                status: Status {
123                    phase: pvc
124                        .status
125                        .as_ref()
126                        .and_then(|s| s.phase.as_deref())
127                        .and_then(Phase::new),
128                    capacity,
129                    current_volume_attributes_class_name: pvc
130                        .status
131                        .as_ref()
132                        .and_then(|s| s.current_volume_attributes_class_name.as_deref()),
133                },
134                volume_name: pvc.spec.as_ref().and_then(|s| s.volume_name.as_deref()),
135            };
136            out.claims.insert(claim_name.to_string(), claim);
137        }
138
139        if out.claims.is_empty() {
140            return None;
141        }
142
143        Some(out)
144    }
145
146    /// Given an [`Pod`] reference, get all of the claim names it
147    /// references in its specification.
148    ///
149    /// Only PVCs are kept, any other kind of volume is discarded.
150    pub fn pod_pvc_claim_names(pod: &Pod) -> impl Iterator<Item = &str> {
151        pod.spec
152            .as_ref()
153            .and_then(|p| p.volumes.as_ref())
154            .into_iter()
155            .flatten()
156            .filter_map(|v| v.persistent_volume_claim.as_ref())
157            .map(|s| s.claim_name.as_str())
158    }
159
160    /// Given a workload, get all of the claim names it references in its Pods'
161    /// specifications.
162    ///
163    /// Only PVCs are kept, any other kind of volume is discarded.
164    ///
165    /// This is a convenience wrapper over [`Self::pod_pvc_claim_names()`].
166    pub fn workload_pvc_claim_names<'p>(
167        pods: impl IntoIterator<Item = &'p Arc<Pod>>,
168    ) -> HashSet<&'p str> {
169        pods.into_iter()
170            .flat_map(|p| Self::pod_pvc_claim_names(p))
171            .collect()
172    }
173}

187/// Live usage data for PVCs. (`kube_pvc_volumes_v1`)
188#[derive(Debug, Default, Serialize)]
189pub(crate) struct KubePvcVolumesV1<'a> {
190    volumes: BTreeMap<String, AttachedVolume<'a>>,
191}
192
193impl<'a> KubePvcVolumesV1<'a> {
194    /// Uses data from the given [`MetricTables`] to provide real-time usage
195    /// data for PVC volumes, in contrast to [`KubePvcV1::from_claim_names()`]
196    /// which uses only API data and does not provide usage data.
197    ///
198    /// Returns None if no claim has volume stats.
199    pub fn from_claim_names(
200        metric_tables: &'a MetricTables,
201        namespace: &'a str,
202        claim_names: impl IntoIterator<Item = &'a str>,
203    ) -> Option<KubePvcVolumesV1<'a>> {
204        let mut out = Self::default();
205        for claim_name in claim_names {
206            let Some(sample) = metric_tables.pvc_volume(namespace, claim_name) else {
207                continue;
208            };
209            let volume = AttachedVolume {
210                capacity: sample.capacity_bytes as f64,
211                free: sample.available_bytes as f64,
212                persistent_volume_claim: claim_name,
213                namespace,
214            };
215            out.volumes.insert(claim_name.to_string(), volume);
216        }
217
218        if out.volumes.is_empty() {
219            return None;
220        }
221
222        Some(out)
223    }
224}

243/// Persistent volumes. (`kube_pvc_pvs_v1`)
244///
245/// Mainly used in the details section of the PVC services.
246#[derive(Debug, Default, Serialize)]
247pub(crate) struct KubePvcPvsV1<'a> {
248    volumes: BTreeMap<String, PersistentVolume<'a>>,
249}
250
251impl<'a> KubePvcPvsV1<'a> {
252    /// Uses the PV index to collect information about the PVs that back the
253    /// PVCs provided by the claim names given.
254    ///
255    /// Returns None if no claims have a backing PV.
256    pub fn from_claim_names(
257        indexes: &'a Indexes,
258        namespace: &'a str,
259        claim_names: impl IntoIterator<Item = &'a str>,
260    ) -> Option<KubePvcPvsV1<'a>> {
261        let mut out = Self::default();
262        for claim_name in claim_names {
263            let Some(pvc) = indexes.pvc(namespace, claim_name) else {
264                continue;
265            };
266
267            let Some(volume_name) = pvc.spec.as_ref().and_then(|s| s.volume_name.as_deref()) else {
268                continue;
269            };
270
271            let Some(pv) = indexes.pv(volume_name) else {
272                continue;
273            };
274
275            let access_modes: &'a [String] = pv
276                .spec
277                .as_ref()
278                .and_then(|spec| spec.access_modes.as_deref())
279                .unwrap_or(&[]);
280
281            let storage_class_name = pv
282                .spec
283                .as_ref()
284                .and_then(|s| s.storage_class_name.as_deref());
285
286            let volume_mode = pv
287                .spec
288                .as_ref()
289                .and_then(|s| s.volume_mode.as_deref())
290                .unwrap_or("Filesystem");
291
292            let pv_spec = PersistentVolumeSpec {
293                access_modes,
294                storage_class_name,
295                volume_mode,
296            };
297
298            let persistent_volume = PersistentVolume {
299                name: volume_name,
300                spec: pv_spec,
301            };
302
303            out.volumes
304                .insert(volume_name.to_string(), persistent_volume);
305        }
306
307        if out.volumes.is_empty() {
308            return None;
309        }
310
311        Some(out)
312    }
313}

329    fn pvc_indexes() -> Indexes {
330        let pvc1 = pvc("pvc-1");
331        let pvc2 = pvc("pvc-2");
332        let mut indexes = Indexes::default();
333        for pvc in [pvc1, pvc2] {
334            indexes
335                .pvcs
336                .entry(pvc.metadata.namespace.clone().unwrap())
337                .or_default()
338                .insert(pvc.metadata.name.clone().unwrap(), pvc.into());
339        }
340        indexes
341    }

343    #[test]
344    fn kube_pvc_v1() {
345        let indexes = pvc_indexes();
346
347        // All claim names match, section render
348        insta::assert_json_snapshot!(KubePvcV1::from_claim_names(
349            &indexes,
350            "really-cool-namespace",
351            ["pvc-1", "pvc-2"]
352        ));
353
354        // If no claim names match, do not produce a section at all
355        assert_matches!(
356            KubePvcV1::from_claim_names(&indexes, "really-cool-namespace", ["bad", "worse"]),
357            None
358        );
359
360        // But if one does, then do produce a section
361        assert_eq!(
362            KubePvcV1::from_claim_names(&indexes, "really-cool-namespace", ["bad", "pvc-1"])
363                .unwrap()
364                .claims
365                .len(),
366            1
367        );
368
369        // Order of claim names does not matter
370        assert_eq!(
371            KubePvcV1::from_claim_names(&indexes, "really-cool-namespace", ["pvc-1", "bad"])
372                .unwrap()
373                .claims
374                .len(),
375            1
376        );
377
378        // If everything is in another namespace, do not produce a section
379        assert_matches!(
380            KubePvcV1::from_claim_names(&indexes, "ANOTHER-namespace", ["pvc-1", "pvc-2"]),
381            None
382        );
383    }
