16/// Many hosts contain aggregations ("roll-ups") of resource and performance
17/// data. These are so common among the piggyback hosts, that we abstract it
18/// out to a trait that the hosts can implement to get free default
19/// implementations of the methods, provided they supply the pod-set to
20/// aggregate over.
21///
22/// The performance methods filter to pods in `Running` phase (via
23/// [`crate::snapshot::metric_tables::MetricTables::aggregate()`]).
24///
25/// Thus, every "aggregation host" can simply implement [`Self::pods()`] and
26/// [`Self::snapshot()`] and get the rest of the roll-up logic for free.
27pub(crate) trait AggregationHost {
28    /// A simple accessor for the current [`Snapshot`].
29    fn snapshot(&self) -> &Snapshot;
30
31    /// The pod-set to aggregate over.
32    fn pods(&self) -> impl Iterator<Item = &Arc<v1::Pod>>;
33
34    fn cpu_resources(&self) -> KubeCpuResourcesV1 {
35        let ra = self
36            .pods()
37            .map(|p| ResourceAccumulator::from_pod(p, ResourceAxis::Cpu))
38            .fold(ResourceAccumulator::default(), Add::add);
39        KubeCpuResourcesV1(ra)
40    }
41
42    fn memory_resources(&self) -> KubeMemoryResourcesV1 {
43        let ra = self
44            .pods()
45            .map(|p| ResourceAccumulator::from_pod(p, ResourceAxis::Memory))
46            .fold(ResourceAccumulator::default(), Add::add);
47        KubeMemoryResourcesV1(ra)
48    }
49
50    fn pod_resources(&self) -> KubePodResourcesV1<'_> {
51        KubePodResourcesV1::from_pods(self.pods().map(Arc::as_ref))
52    }
53
54    fn cpu_performance(&self) -> Option<KubePerformanceCpuV1> {
55        Some(KubePerformanceCpuV1::new(
56            self.snapshot()
57                .metrics
58                .aggregate(self.pods())
59                .cpu_usage_nano_cores?,
60        ))
61    }
62
63    fn memory_performance(&self) -> Option<KubePerformanceMemoryV1> {
64        Some(KubePerformanceMemoryV1::new(
65            self.snapshot()
66                .metrics
67                .aggregate(self.pods())
68                .memory_working_set_bytes?,
69        ))
70    }
71
72    fn swap_performance(&self) -> Option<KubePerformanceMemorySwapV1> {
73        Some(KubePerformanceMemorySwapV1::new(
74            self.snapshot()
75                .metrics
76                .aggregate(self.pods())
77                .swap_usage_bytes?,
78        ))
79    }
80
81    /// Emit performance and resource utilization sections for the pod set
82    /// returned by [`Self::pods()`].
83    ///
84    /// Note that this does *not* include [`KubePodResourcesV1`], because we do
85    /// not emit that section for pods. So [`crate::piggyback::pod::Pod`] emits
86    /// from this function, and every *other* `AggregationHost` calls
87    /// [`Self::aggregation_sections()`] instead, which includes this subset along
88    /// with [`KubePodResourcesV1`].
89    fn performance_and_resource_util_sections(
90        &self,
91        me: &str,
92    ) -> Vec<Result<WriteableSection, SectionError>> {
93        let mut out = vec![
94            WriteableSection::of(me, &self.cpu_resources()),
95            WriteableSection::of(me, &self.memory_resources()),
96        ];
97        if let Some(cpu_perf) = &self.cpu_performance() {
98            out.push(WriteableSection::of(me, cpu_perf));
99        }
100        if let Some(mem_perf) = &self.memory_performance() {
101            out.push(WriteableSection::of(me, mem_perf));
102        }
103        if let Some(swap_perf) = &self.swap_performance() {
104            out.push(WriteableSection::of(me, swap_perf));
105        }
106        out
107    }
108
109    /// Emit all possible resources and performance sections for the pod set
110    /// returned by [`Self::pods()`]. This includes everything that
111    /// [`Self::performance_and_resource_util_sections()`] returns plus
112    /// [`KubePodResourcesV1`].
113    fn aggregation_sections(&self, me: &str) -> Vec<Result<WriteableSection, SectionError>> {
114        let mut out = self.performance_and_resource_util_sections(me);
115        out.push(WriteableSection::of(me, &self.pod_resources()));
116        out
117    }
118
119    /// Emit PVC sections for claims referenced by this host's Pods.
120    ///
121    /// The legacy agent emits these sections on Pod, Deployment, DaemonSet,
122    /// and StatefulSet hosts when PVC monitoring is enabled. This is an output
123    /// placement compatibility rule, not an inherent data limitation:
124    /// Namespace and CronJob hosts could also provide the data, but the legacy
125    /// agent does not associate PVC services with them. Cluster and Node hosts
126    /// additionally have no single namespace in which to resolve a claim.
127    /// Therefore this deliberately stays separate from
128    /// [`Self::aggregation_sections()`], whose sections apply to every
129    /// aggregation host.
130    fn pvc_sections(
131        &self,
132        me: &str,
133        namespace: &str,
134    ) -> Vec<Result<WriteableSection, SectionError>> {
135        let claim_names = KubePvcV1::workload_pvc_claim_names(self.pods());
136        let mut out = Vec::new();
137
138        if let Some(section) = KubePvcV1::from_claim_names(
139            &self.snapshot().indexes,
140            namespace,
141            claim_names.iter().copied(),
142        ) {
143            out.push(WriteableSection::of(me, &section));
144        }
145        if let Some(section) = KubePvcVolumesV1::from_claim_names(
146            &self.snapshot().metrics,
147            namespace,
148            claim_names.iter().copied(),
149        ) {
150            out.push(WriteableSection::of(me, &section));
151        }
152        if let Some(section) = KubePvcPvsV1::from_claim_names(
153            &self.snapshot().indexes,
154            namespace,
155            claim_names.iter().copied(),
156        ) {
157            out.push(WriteableSection::of(me, &section));
158        }
159        out
160    }
161}
