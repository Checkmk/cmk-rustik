15pub struct Cluster<'a> {
16    snapshot: &'a Snapshot,
17    settings: &'a HostSettings,
18    aggregation_nodes: Vec<&'a Arc<Node>>,
19    aggregation_pods: Vec<&'a Arc<Pod>>,
20}
21
22impl<'a> Cluster<'a> {
23    /// Create a new cluster piggyback host instance.
24    ///
25    /// Importantly, nodes considered here are potentially filtered by user
26    /// preference and the node role patterns to exclude are specified as a CLI
27    /// argument and land in the [`HostSettings`] `excluded_node_role_patterns`
28    /// field.
29    ///
30    /// This impacts both the nodes that are considered here and pod
31    /// aggregations.
32    ///
33    /// For purposes of aggregation/roll-up, pods are only considered if their
34    /// node name is `None` or if their node is not excluded.
35    pub fn new(snapshot: &'a Snapshot, settings: &'a HostSettings) -> Cluster<'a> {
36        let aggregation_nodes: Vec<&Arc<Node>> =
37            Self::aggregation_nodes(&snapshot.stores.nodes, settings);
38
39        // if the node name of the pod is None, keep it. If it is Some, check if it is in aggregation_nodes.
40        let aggregation_pods: Vec<&Arc<Pod>> =
41            Self::aggregation_pods(&snapshot.stores.pods, &aggregation_nodes);
42
43        Cluster {
44            snapshot,
45            settings,
46            aggregation_nodes,
47            aggregation_pods,
48        }
49    }
50
51    fn aggregation_nodes(nodes: &'a [Arc<Node>], settings: &'a HostSettings) -> Vec<&'a Arc<Node>> {
52        nodes
53            .iter()
54            .filter(|n| !settings.is_node_excluded(n))
55            .collect()
56    }
57
58    fn aggregation_pods(
59        pods: &'a [Arc<Pod>],
60        aggregation_nodes: &[&'a Arc<Node>],
61    ) -> Vec<&'a Arc<Pod>> {
62        let aggregation_node_names: HashSet<&str> = aggregation_nodes
63            .iter()
64            .filter_map(|n| n.metadata.name.as_deref())
65            .collect();
66        pods.iter()
67            .filter(
68                |p| match p.spec.as_ref().and_then(|s| s.node_name.as_deref()) {
69                    None => true,
70                    Some(name) => aggregation_node_names.contains(&name),
71                },
72            )
73            .collect()
74    }
75}

77impl AggregationHost for Cluster<'_> {
78    fn snapshot(&self) -> &Snapshot {
79        self.snapshot
80    }
81
82    fn pods(&self) -> impl Iterator<Item = &Arc<Pod>> {
83        self.aggregation_pods.iter().copied()
84    }
85}

87impl PiggybackHost for Cluster<'_> {
88    fn metadata(&self) -> Option<&ObjectMeta> {
89        None
90    }
91
92    fn kind(&self) -> &str {
93        "cluster"
94    }
95
96    fn emit(&self) -> Vec<Result<WriteableSection, SectionError>> {
97        let me = "";
98        let mut out = Vec::new();
99        out.push(WriteableSection::of(
100            me,
101            &KubeClusterInfoV1::from_host_settings(self.settings),
102        ));
103        out.push(WriteableSection::of(
104            me,
105            // Deliberately every node, not just the aggregation nodes.
106            &KubeNodeCountV1::from_nodes(&self.snapshot.stores.nodes),
107        ));
108        out.extend(self.aggregation_sections(me));
109        out.push(WriteableSection::of(
110            me,
111            &KubeAllocatablePodsV1::from_nodes(
112                self.aggregation_nodes.iter().copied().map(Arc::as_ref),
113            ),
114        ));
115        out.push(WriteableSection::of(
116            me,
117            &KubeAllocatableCpuResourceV1::from_nodes(
118                self.aggregation_nodes.iter().copied().map(Arc::as_ref),
119            ),
120        ));
121        out.push(WriteableSection::of(
122            me,
123            &KubeAllocatableMemoryResourceV1::from_nodes(
124                self.aggregation_nodes.iter().copied().map(Arc::as_ref),
125            ),
126        ));
127        out.push(WriteableSection::of(
128            me,
129            &KubeRustikHealthV1::from_self_health(&self.snapshot.self_health),
130        ));
131        if let Some(kube_cluster_details_v1) =
132            KubeClusterDetailsV1::new(&self.snapshot.api_health_update)
133        {
134            out.push(WriteableSection::of(me, &kube_cluster_details_v1));
135        }
136
137        out
138    }
139}

158    #[tokio::test]
159    async fn test_node_count_covers_nodes_excluded_from_aggregation() {
160        let settings = HostSettings {
161            excluded_node_role_patterns: vec![Regex::new("control-plane").unwrap()],
162            ..host_settings()
163        };
164        let snapshot = snapshot_with_nodes(vec![
165            Arc::new(node_with_roles("control-1", &["control-plane"])),
166            Arc::new(node_with_roles("worker-1", &["worker"])),
167        ]);
168
169        let cluster = Cluster::new(&snapshot, &settings);
170        assert_eq!(cluster.aggregation_nodes.len(), 1, "control plane excluded");
171
172        let body = cluster
173            .emit()
174            .into_iter()
175            .filter_map(Result::ok)
176            .find_map(|section| match section.body {
177                SectionBody::Json { name, body } if name == KubeNodeCountV1::NAME => Some(body),
178                _ => None,
179            })
180            .expect("expected a kube_node_count_v1 section");
181        let section: serde_json::Value = serde_json::from_str(&body).unwrap();
182
183        let counted = section["nodes"].as_array().expect("nodes is an array");
184        assert_eq!(counted.len(), 2, "both nodes are counted");
185        assert_eq!(counted[0]["roles"][0], "control-plane");
186        assert_eq!(counted[1]["roles"][0], "worker");
187    }

189    #[test]
190    /// Test that [`Cluster::aggregation_pods()`] includes pods only running on
191    /// non-excluded nodes.
192    fn test_aggregation_pods() {
193        let nodes = vec![Arc::new(node("worker-1"))];
194        let aggregation_nodes: Vec<&Arc<Node>> = nodes.iter().collect();
195
196        let pods = vec![
197            Arc::new(pod("on-worker", Some("worker-1"))), // node is included
198            Arc::new(pod("on-control", Some("control-plane"))), // node is dropped
199            Arc::new(pod("unscheduled", None)),           // node is included
200        ];
201
202        let result = Cluster::aggregation_pods(&pods, &aggregation_nodes);
203        let names: Vec<&str> = result
204            .iter()
205            .filter_map(|p| p.metadata.name.as_deref())
206            .collect();
207        assert_eq!(names, vec!["on-worker", "unscheduled"]);
208    }
