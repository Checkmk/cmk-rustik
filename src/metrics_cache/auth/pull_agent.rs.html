1//! Authentication machinery for the pull special agent to work.
2//!
3//! The pull agent will send us `Authorization: Bearer <shared_secret>`
4//!
5//! The value of the shared secret is passed in via an environment variable,
6//! `CMK_PULL_SHARED_SECRET`. In a typical deployment, it lives in a Kubernetes
7//! secret and the env var is bound from that. It is read by clap in
8//! [`crate::cli_args::CliArgs`] and can also be overridden via a CLI flag,
9//! though this is unsafe (since the process information would leak it) and we
10//! do not do this in the Helm chart.
11
12use axum::extract::{Request, State};
13use axum::http::{StatusCode, header};
14use axum::middleware::Next;
15use axum::response::Response;
16use subtle::ConstantTimeEq;
17
18/// Configuration for the pull-agent middleware.
19///
20/// Right now, includes the shared secret; in theory in the future could also
21/// include things like an IP whitelist.
22#[derive(Clone)]
23pub struct PullAgentMiddlewareConfig {
24    /// Defines if authentication should be required for pull endpoints at all
25    pub auth_enabled: bool,
26    /// The shared secret that we expect to be sent by the Checkmk special agent
27    /// for authentication. We expect to find this in the `Authorization` header
28    /// after `Bearer`
29    pub shared_secret: Option<String>,
30}
31
32impl Default for PullAgentMiddlewareConfig {
33    /// Defaults closed: auth enabled with no configured secret, so
34    /// [`authorized`] rejects every request. This is only meant as a
35    /// placeholder for tests that don't exercise pull-agent auth at all; if it
36    /// were ever reached in real code, it fails safe instead of leaving the
37    /// endpoint open.
38    fn default() -> Self {
39        Self {
40            auth_enabled: true,
41            shared_secret: None,
42        }
43    }
44}
45
46/// Perform authentication, specifically for pull-agent endpoints.
47///
48/// If `config.shared_secret` is `None` or empty, we _REJECT_ all requests.
49/// `None` means pull mode is not configured at all; an empty value is what a
50/// misconfigured secret (wrong key, empty env var) produces, so it must never
51/// mean "public". The only way to get an unauthenticated pull endpoint is the
52/// explicit `--disable-pull-authentication` flag (`auth_enabled: false`).
53///
54/// Otherwise we only permit the request if its `Authorization: Bearer <...>`
55/// value matches the shared secret value.
56pub(crate) async fn authenticate(
57    State(config): State<PullAgentMiddlewareConfig>,
58    request: Request,
59    next: Next,
60) -> Result<Response, StatusCode> {
61    let authorization = request
62        .headers()
63        .get(header::AUTHORIZATION)
64        .and_then(|header| header.to_str().ok());
65    if authorized(&config, authorization) {
66        Ok(next.run(request).await)
67    } else {
68        Err(StatusCode::UNAUTHORIZED)
69    }
70}
71
72/// The pure authentication decision: is a request carrying this
73/// `Authorization` header value allowed through under this config?
74fn authorized(config: &PullAgentMiddlewareConfig, authorization: Option<&str>) -> bool {
75    if !config.auth_enabled {
76        return true;
77    }
78    match config.shared_secret.as_deref() {
79        // Unconfigured or misconfigured-empty: fail closed.
80        None | Some("") => false,
81        Some(configured_secret) => {
82            let Some(received_secret) =
83                authorization.and_then(|value| value.strip_prefix("Bearer "))
84            else {
85                return false;
86            };
87            // Constant-time comparison so response timing does not leak
88            // how much of the secret matched.
89            received_secret
90                .as_bytes()
91                .ct_eq(configured_secret.as_bytes())
92                .into()
93        }
94    }
95}

97#[cfg(test)]
98mod tests {
99    use super::*;
100
101    fn config(auth_enabled: bool, shared_secret: Option<&str>) -> PullAgentMiddlewareConfig {
102        PullAgentMiddlewareConfig {
103            auth_enabled,
104            shared_secret: shared_secret.map(String::from),
105        }
106    }
107
108    #[test]
109    fn authorized_policies() {
110        // The happy path, and the only accepted header shape.
111        assert!(authorized(&config(true, Some("s3c")), Some("Bearer s3c")));
112        assert!(!authorized(&config(true, Some("s3c")), Some("Bearer nope")));
113        assert!(!authorized(&config(true, Some("s3c")), Some("Secret s3c")));
114        assert!(!authorized(&config(true, Some("s3c")), Some("bearer s3c")));
115        assert!(!authorized(&config(true, Some("s3c")), Some("s3c")));
116        assert!(!authorized(&config(true, Some("s3c")), Some("")));
117        assert!(!authorized(&config(true, Some("s3c")), None));
118
119        // No secret configured: fail closed
120        assert!(!authorized(&config(true, None), Some("Bearer s3c")));
121
122        // Empty secret never matches, not even an empty presented one.
123        assert!(!authorized(&config(true, Some("")), Some("Bearer ")));
124
125        // Auth explicitly disabled: everything passes.
126        assert!(authorized(&config(false, None), None));
127    }
128}
