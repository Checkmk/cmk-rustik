21// Used for the size of the various metrics-fetcher caches.
22const MAX_SUPPORTED_KUBERNETES_NODES: u64 = 5000;

24#[derive(Clone)]
25pub struct AppState<V: TokenValidator> {
26    pub client: V,
27    pub stores: Stores,
28    pub reader_allowlist: Vec<String>,
29    pub writer_allowlist: Vec<String>,
30    pub kubelet_stats_summary_cache: Cache<String, Arc<MetricsFetcherIngestion<StatsSummary>>>,
31    pub kubelet_health_cache: Cache<String, Arc<MetricsFetcherIngestion<KubeletHealth>>>,
32    pub system_agent_cache: Cache<String, Arc<MetricsFetcherIngestion<SystemAgentOutput>>>,
33    pub host_settings: Arc<HostSettings>,
34    pub api_health_receiver: Receiver<ApiHealthUpdate>,
35    pub metrics_fetcher_daemonset: Option<MetricsFetcherDaemonSet>,
36}

38impl AppState<Client> {
39    pub async fn new(
40        args: &CliArgs,
41        tasks: &mut JoinSet<()>,
42        api_health_receiver: Receiver<ApiHealthUpdate>,
43    ) -> Result<Self> {
44        let client = Self::kube_client(args.connect_timeout, args.read_timeout).await?;
45        let watcher_client = Self::kube_watcher_client(args.connect_timeout).await?;
46        let cluster_version = client
47            .apiserver_version()
48            .await
49            .map_err(Error::KubeApiServerVersion)?
50            .git_version;
51        let host_settings = HostSettings {
52            cluster_name: args.cluster_name.clone(),
53            cluster_host_name: args.cluster_host_name.clone(),
54            annotation_key_pattern: AnnotationKeyPattern::new(
55                args.import_all_annotations,
56                args.annotation_key_pattern.clone(),
57            ),
58            excluded_node_role_patterns: args.excluded_node_role_patterns.clone(),
59            namespace_filter: NamespaceFilter::new(
60                args.namespace_include_patterns.clone(),
61                args.namespace_exclude_patterns.clone(),
62            ),
63            always_emitted: AlwaysEmitted::from_cli_args(args),
64            include_cronjob_pods: args.include_cronjob_pods,
65            emit_pvc_sections: args.all_pvcs,
66            cluster_version,
67        };
68        let stores = Stores::spawn(watcher_client, tasks);
69
70        let state = Self {
71            client,
72            stores,
73            reader_allowlist: args.reader_allowlist.clone(),
74            writer_allowlist: args.writer_allowlist.clone(),
75            kubelet_stats_summary_cache: Cache::builder()
76                .time_to_live(args.kubelet_stats_cache_ttl)
77                .max_capacity(MAX_SUPPORTED_KUBERNETES_NODES)
78                .build(),
79            kubelet_health_cache: Cache::builder()
80                .time_to_live(args.kubelet_health_cache_ttl)
81                .max_capacity(MAX_SUPPORTED_KUBERNETES_NODES)
82                .build(),
83            system_agent_cache: Cache::builder()
84                .time_to_live(args.system_agent_cache_ttl)
85                .max_capacity(MAX_SUPPORTED_KUBERNETES_NODES)
86                .build(),
87            host_settings: host_settings.into(),
88            api_health_receiver,
89            metrics_fetcher_daemonset: args
90                .namespace
91                .clone()
92                .zip(args.metrics_fetcher_daemonset_name.clone())
93                .map(|(namespace, name)| MetricsFetcherDaemonSet { namespace, name }),
94        };
95        Ok(state)
96    }

98    /// Build a Kubernetes client for general use (token reviews, etc.).
99    async fn kube_client(connect_timeout: Duration, read_timeout: Duration) -> Result<Client> {
100        let mut config = kube::Config::infer().await?;
101        config.connect_timeout = Some(connect_timeout);
102        config.read_timeout = Some(read_timeout);
103
104        Ok(Client::try_from(config)?)
105    }
106
107    /// Build a Kubernetes client suitable for watch streams. No read timeout —
108    /// watch connections are long-lived and idle between events, so a read timeout
109    /// would kill them.
110    async fn kube_watcher_client(connect_timeout: Duration) -> Result<Client> {
111        let mut config = kube::Config::infer().await?;
112        config.connect_timeout = Some(connect_timeout);
113
114        Ok(Client::try_from(config)?)
115    }
116}

118impl<V: TokenValidator> AppState<V> {
119    /// Create a [`Snapshot`] from the current [`AppState`].
120    pub fn snapshot(&self) -> Snapshot {
121        Snapshot::new(
122            self.stores.clone(),
123            self.kubelet_stats_summary_cache.clone(),
124            self.kubelet_health_cache.clone(),
125            self.system_agent_cache.clone(),
126            self.api_health_receiver.clone(),
127            self.metrics_fetcher_daemonset.as_ref(),
128        )
129    }
130}
