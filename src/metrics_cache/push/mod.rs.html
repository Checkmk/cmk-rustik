39/// Generate, compress, and push sections from the current state.
40///
41/// This is what actually generates the sections, compresses them, and pushes
42/// them to Checkmk using the push client. It runs once per push interval,
43/// called via [`push_loop()`], and builds a [`crate::snapshot::Snapshot`],
44/// emits all sections, zlib-compresses them, and pushes via the
45/// [`CheckmkPushClient`].
46async fn push_cycle(
47    client: &CheckmkPushClient,
48    state: &AppState<impl TokenValidator>,
49) -> anyhow::Result<()> {
50    let snap = state.snapshot();
51    let sections = emit_all(&snap, &state.host_settings);
52    let mut encoder = ZlibEncoder::new(Vec::new(), Compression::default());
53    frame(&mut encoder, sections).context("framing push data")?;
54    let encoded = encoder.finish().context("finishing zlib encoding")?;
55    client.push_section_data(encoded).await?;
56    Ok(())
57}

59async fn renew_certificate_if_needed(
60    client: &mut CheckmkPushClient,
61    registration: &CheckmkPushRegistration<'_>,
62    renewal_threshold: Duration,
63) {
64    if !client.begin_renewal_attempt(renewal_threshold) {
65        return;
66    }
67    match registration.renew(client).await {
68        Ok(replacement) => client.replace_identity(replacement),
69        Err(error) => {
70            error!(?error, "Failed to renew push agent certificate");
71        }
72    }
73}

75/// Run the push loop forever.
76///
77/// Every push interval, [`push_cycle()`] is called to generate sections and
78/// send them to the Checkmk server.
79///
80/// Waits until all stores are ready before doing its initial push and looping.
81///
82/// After each push attempt, renews the client certificate when needed.
83pub async fn push_loop(
84    mut client: CheckmkPushClient,
85    registration: CheckmkPushRegistration<'_>,
86    state: AppState<impl TokenValidator>,
87    push_interval: Duration,
88    certificate_renewal_threshold: Duration,
89) -> Result<(), WriterDropped> {
90    state.stores.wait_until_all_ready().await?;
91    let mut interval = time::interval(push_interval);
92    loop {
93        interval.tick().await; // note: The very first tick() is no-op
94        match push_cycle(&client, &state).await {
95            Ok(_) => debug!("Successfully pushed metrics to Checkmk server"),
96            Err(e) => error!(error = ?e, "Failed to push metrics to Checkmk server"),
97        }
98        renew_certificate_if_needed(&mut client, &registration, certificate_renewal_threshold)
99            .await;
100    }
101}
