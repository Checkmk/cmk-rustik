1//! This module contains the push client, to push section data into a Checkmk
2//! server.
3
4use k8s_openapi::api::core::v1::Secret;
5use reqwest::multipart;
6use tracing::{error, trace};
7use uuid::Uuid;
8
9use crate::error::Result;
10use crate::push::Error;
11use crate::push::renew::RenewalState;
12use crate::push::server_cert_verifier;
13
14/// A client to push section data into a Checkmk server.
15///
16/// It assumes an existing certificate and key. To generate them and register
17/// with the Checkmk server, see [`crate::push::register`].
18pub struct CheckmkPushClient {
19    pub client: reqwest::Client,
20    pub push_url: String,
21    pub(super) renewal: RenewalState,
22}
23
24fn secret_value(secret: &Secret, key: &str, description: &str) -> Result<String> {
25    secret
26        .data
27        .as_ref()
28        .and_then(|data| data.get(key))
29        .map(|bytes| String::from_utf8_lossy(&bytes.0).into_owned())
30        .ok_or_else(|| {
31            error!("{description} missing from secret");
32            Error::PushMode(format!("{description} missing from secret")).into()
33        })
34}
35
36impl CheckmkPushClient {
37    pub fn from_secret(base_url: &str, secret: &Secret) -> Result<Self> {
38        trace!("Creating CheckmkPushClient from secret");
39        let key = secret_value(secret, "private_key", "private key")?;
40        let root_cert = secret_value(secret, "root_cert", "root certificate")?;
41        let agent_cert = secret_value(secret, "agent_cert", "agent certificate")?;
42        let uuid = secret_value(secret, "uuid", "uuid")?;
43        let uuid = Uuid::parse_str(&uuid).map_err(|error| {
44            error!(?error, "invalid uuid in push certificate secret");
45            Error::PushMode("invalid uuid in push certificate secret".to_string())
46        })?;
47        let tls = server_cert_verifier::client_config(&root_cert, &agent_cert, &key).map_err(
48            |error| {
49                error!(error = ?error, "failed to configure push-mode TLS");
50                Error::TlsClientConfig(error)
51            },
52        )?;
53        let renewal = RenewalState::new(base_url, uuid, &tls);
54        let client = reqwest::Client::builder()
55            .use_preconfigured_tls(tls.config)
56            .build()
57            .map_err(|e| {
58                error!(error = ?e, "failed to build push-mode client");
59                Error::PushMode("failed to build push-mode client".to_string())
60            })?;
61        let push_url = format!(
62            "{}/agent-receiver/agent_data/{}",
63            base_url.trim_end_matches('/'),
64            uuid
65        );
66        Ok(Self {
67            client,
68            push_url,
69            renewal,
70        })
71    }
72
73    pub(super) async fn push_section_data(&self, section_data: Vec<u8>) -> Result<()> {
74        let part = multipart::Part::bytes(section_data)
75            .file_name("agent_data")
76            .mime_str("application/octet-stream")
77            .map_err(|e| {
78                error!(error = ?e, "failed to create multipart part for section data");
79                Error::PushMode("failed to create multipart part for section data".to_string())
80            })?;
81        let form = multipart::Form::new().part("monitoring_data", part);
82        let response = self
83            .client
84            .post(&self.push_url)
85            .header("compression", "zlib")
86            .multipart(form)
87            .send()
88            .await
89            .map_err(|e| {
90                error!(error = ?e, "failed to send section data to Checkmk server");
91                Error::PushMode("failed to send section data to Checkmk server".to_string())
92            })?;
93        if !response.status().is_success() {
94            let status = response.status();
95            let body = response
96                .text()
97                .await
98                .unwrap_or_else(|_| "<failed to read body>".to_string());
99            error!(%status, %body, "Failed to push section data to Checkmk server");
100            return Err(Error::PushMode(format!(
101                "Failed to push section data to Checkmk server: {}",
102                status
103            ))
104            .into());
105        }
106        Ok(())
107    }
108}
