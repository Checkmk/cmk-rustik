1//! This module contains the OTel client, to export OTLP metrics to an OTel
2//! collector.
3
4use opentelemetry_proto::tonic::collector::metrics::v1::ExportMetricsServiceRequest;
5use prost::Message;
6
7use crate::error::Result;
8use crate::otel::Error;
9
10/// A client to export OTLP metrics to an OTel collector over OTLP/http.
11pub struct OtelClient {
12    client: reqwest::Client,
13    export_url: String,
14}
15
16impl OtelClient {
17    pub fn new(endpoint: &str) -> Self {
18        Self {
19            client: reqwest::Client::new(),
20            export_url: format!("{}/v1/metrics", endpoint.trim_end_matches('/')),
21        }
22    }
23
24    /// Send one export request to the collector.
25    ///
26    /// Errors on connection failure or a non-success HTTP status; the caller
27    /// decides whether that is fatal (for the export loop it is not).
28    pub(super) async fn export(&self, request: ExportMetricsServiceRequest) -> Result<()> {
29        let response = self
30            .client
31            .post(&self.export_url)
32            .header("content-type", "application/x-protobuf")
33            .body(request.encode_to_vec())
34            .send()
35            .await
36            .map_err(Error::Http)?;
37        let status = response.status();
38        if !status.is_success() {
39            let body = response
40                .text()
41                .await
42                .unwrap_or_else(|_| "<failed to read body>".to_string());
43            return Err(Error::Rejected { status, body }.into());
44        }
45        Ok(())
46    }
47}
