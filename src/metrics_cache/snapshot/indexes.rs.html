1use k8s_openapi::api::core::v1::{PersistentVolume, PersistentVolumeClaim, Pod, ResourceQuota};
2use std::collections::HashMap;
3use std::sync::Arc;
4
5use crate::ingest::reflectors::FrozenStores;
6
7#[derive(Debug, Default)]
8pub struct Indexes {
9    /// Map namespace names to all the pods within that namespace.
10    pub pods_by_namespace: HashMap<String, Vec<Arc<Pod>>>,
11
12    /// Map node names to all pods on that node.
13    pub pods_by_node: HashMap<String, Vec<Arc<Pod>>>,
14
15    /// Persistent Volume Claims, indexed by `pvc = pvcs[namespace][name]`.
16    ///
17    /// Pods reference a PVC by name (and only in the same namespace).
18    pub pvcs: HashMap<String, HashMap<String, Arc<PersistentVolumeClaim>>>,
19
20    /// Persistent volumes, by name.
21    pub pvs: HashMap<String, Arc<PersistentVolume>>,
22
23    /// ResourceQuotas by namespace. We intentionally do not support multiple
24    /// quotas per namespace currently.
25    pub resource_quota_by_namespace: HashMap<String, Arc<ResourceQuota>>,
26}
27
28impl Indexes {
29    pub fn from_frozen_stores(stores: &FrozenStores) -> Self {
30        Self {
31            pods_by_namespace: Self::get_pods_by_namespace(stores),
32            pods_by_node: Self::get_pods_by_node(stores),
33            pvcs: Self::pvcs_by_ns_and_name(stores),
34            pvs: Self::get_pvs_by_name(stores),
35            resource_quota_by_namespace: Self::get_resource_quota_by_namespace(stores),
36        }
37    }
38
39    /// Iterate the PVCs in the store and map them out to be keyed on their
40    /// namespace and name. O(n) over the PVCs in the store.
41    fn pvcs_by_ns_and_name(
42        stores: &FrozenStores,
43    ) -> HashMap<String, HashMap<String, Arc<PersistentVolumeClaim>>> {
44        let mut out: HashMap<String, HashMap<String, Arc<PersistentVolumeClaim>>> = HashMap::new();
45        for pvc in &stores.persistent_volume_claims {
46            let Some(namespace) = pvc.metadata.namespace.clone() else {
47                continue;
48            };
49            let Some(name) = pvc.metadata.name.clone() else {
50                continue;
51            };
52            out.entry(namespace).or_default().insert(name, pvc.clone());
53        }
54        out
55    }
56
57    /// Given a namespace and a PVC name, try to find the PVC. O(1).
58    pub fn pvc(&self, namespace: &str, name: &str) -> Option<&PersistentVolumeClaim> {
59        self.pvcs.get(namespace)?.get(name).map(Arc::as_ref)
60    }
61
62    /// List all the pods in the given namespace
63    pub fn pods_by_namespace(&self, namespace: &str) -> &[Arc<Pod>] {
64        self.pods_by_namespace
65            .get(namespace)
66            .map_or(&[], Vec::as_slice)
67    }
68
69    /// Compute all the pods associated with each namespace in the snapshot.
70    fn get_pods_by_namespace(stores: &FrozenStores) -> HashMap<String, Vec<Arc<Pod>>> {
71        let mut out: HashMap<String, Vec<Arc<Pod>>> = HashMap::new();
72        for pod in &stores.pods {
73            let Some(namespace) = &pod.metadata.namespace else {
74                continue;
75            };
76            if let Some(namespace_vec) = out.get_mut(namespace) {
77                namespace_vec.push(pod.clone());
78            } else {
79                out.insert(namespace.clone(), vec![pod.clone()]);
80            }
81        }
82        out
83    }
84
85    /// List all the pods scheduled on the given node name
86    pub fn pods_by_node(&self, node: &str) -> &[Arc<Pod>] {
87        self.pods_by_node.get(node).map_or(&[], Vec::as_slice)
88    }
89
90    /// Compute all the pods scheduled on each node in the snapshot.
91    fn get_pods_by_node(stores: &FrozenStores) -> HashMap<String, Vec<Arc<Pod>>> {
92        let mut out: HashMap<String, Vec<Arc<Pod>>> = HashMap::new();
93        for pod in &stores.pods {
94            let Some(node) = pod.spec.as_ref().and_then(|s| s.node_name.as_ref()) else {
95                continue;
96            };
97            out.entry(node.clone()).or_default().push(pod.clone());
98        }
99        out
100    }
101
102    /// Index all PVs by their (cluster-global) name.
103    fn get_pvs_by_name(stores: &FrozenStores) -> HashMap<String, Arc<PersistentVolume>> {
104        let mut out: HashMap<String, Arc<PersistentVolume>> = HashMap::new();
105        for pv in &stores.persistent_volumes {
106            let Some(name) = &pv.metadata.name else {
107                continue;
108            };
109            out.insert(name.clone(), pv.clone());
110        }
111        out
112    }
113
114    /// Get a (cluster-global) PV by name. O(1).
115    pub fn pv(&self, name: &str) -> Option<&PersistentVolume> {
116        self.pvs.get(name).map(Arc::as_ref)
117    }
118
119    /// Index all ResourceQuotas by namespace.
120    fn get_resource_quota_by_namespace(
121        stores: &FrozenStores,
122    ) -> HashMap<String, Arc<ResourceQuota>> {
123        let mut out: HashMap<String, Arc<ResourceQuota>> = HashMap::new();
124        for quota in &stores.resourcequotas {
125            let Some(namespace) = &quota.metadata.namespace else {
126                continue;
127            };
128            // Support only one quota.
129            if out.contains_key(namespace) {
130                continue;
131            }
132            out.insert(namespace.clone(), quota.clone());
133        }
134        out
135    }
136
137    /// Get a Namespace's ResourceQuota if it exists. O(1).
138    pub fn resource_quota(&self, namespace: &str) -> Option<&ResourceQuota> {
139        self.resource_quota_by_namespace
140            .get(namespace)
141            .map(Arc::as_ref)
142    }
143}
