12/// An [`OwnerGraph`] is a structure which allows for lookups of which resource
13/// owns another resource, if any. For example, a Pod might be owned by a
14/// ReplicaSet which might be owned by a Deployment.
15///
16/// `OwnerGraph` owns the structures to perform such lookups and exposes
17/// functions to make use of said structures.
18///
19/// When the graph is created, internally three maps are created:
20///
21/// - The first maps Kubernetes [`Uid`]s of all "controllable" objects that we
22///   monitor (e.g. Pods, ReplicaSets, etc.) to [`OwnerReference`]s. This is a
23///   **direct, non-recursive** mapping: one object to its immediate owner as
24///   reported by Kubernetes. The only requirement is that the owner is listed
25///   as a (_the_) controller for the object.
26///
27/// - The second maps controllers (things that can own other things), by their
28///   [`Uid`]s, to a vector containing the pods that they own. This mapping is
29///   recursive (represents the transitive closure of all the pods owned by the
30///   controller in the key).
31///
32/// - The third maps controllers to their directly owned Jobs.
33#[derive(Debug)]
34pub struct OwnerGraph {
35    pub owner_ref_by_uid: HashMap<Uid, OwnerReference>,
36    pub pods_by_controller: HashMap<Uid, Vec<Arc<Pod>>>,
37    pub jobs_by_controller: HashMap<Uid, Vec<Arc<Job>>>,
38}
39
40impl OwnerGraph {
41    /// Construct an `OwnerGraph` given a [`FrozenStores`] reference.
42    ///
43    /// Outside of testing, this normally only gets called by
44    /// [`crate::snapshot::Snapshot::new()`].
45    pub fn from_frozen_stores(stores: &FrozenStores) -> Self {
46        let owner_ref_by_uid =
47            Self::map_object_uids_to_owner_ref(&stores.pods, &stores.replicasets, &stores.jobs);
48        let pods_by_controller = Self::get_pods_by_controller(&stores.pods, &owner_ref_by_uid);
49        let jobs_by_controller = Self::get_jobs_by_controller(&stores.jobs, &owner_ref_by_uid);
50        OwnerGraph {
51            owner_ref_by_uid,
52            pods_by_controller,
53            jobs_by_controller,
54        }
55    }
56
57    /// Create a map, uid-to-OwnerReference, of objects monitored in the
58    /// [`crate::ingest::reflectors::Stores`]. We take the first owner which
59    /// reports being a controller.
60    ///
61    /// Theory says there should only be at most one per object anyway.
62    /// If one is found, the object's [`Uid`] is used as the key in the map
63    /// and the [`OwnerReference`] is taken as the value.
64    fn map_object_uids_to_owner_ref(
65        pods: &[Arc<Pod>],
66        replicasets: &[Arc<ReplicaSet>],
67        jobs: &[Arc<Job>],
68    ) -> HashMap<Uid, OwnerReference> {
69        let mut map = HashMap::new();
70        for pod in pods {
71            if let Some(owner_controller) = pod
72                .owner_references()
73                .iter()
74                .find(|r| r.controller == Some(true))
75                && let Some(uid) = pod.metadata.uid.clone()
76            {
77                map.insert(Uid(uid.into()), owner_controller.to_owned());
78            }
79        }
80        for rs in replicasets {
81            if let Some(owner_controller) = rs
82                .owner_references()
83                .iter()
84                .find(|r| r.controller == Some(true))
85                && let Some(uid) = rs.metadata.uid.clone()
86            {
87                map.insert(Uid(uid.into()), owner_controller.to_owned());
88            }
89        }
90        for job in jobs {
91            if let Some(owner_controller) = job
92                .owner_references()
93                .iter()
94                .find(|r| r.controller == Some(true))
95                && let Some(uid) = job.metadata.uid.clone()
96            {
97                map.insert(Uid(uid.into()), owner_controller.to_owned());
98            }
99        }
100        map
101    }
102
103    /// Build the inverse index: for each controller UID, the UIDs of all pods
104    /// it transitively owns. Each pod is registered under every ancestor in its
105    /// chain, so, for example, a Deployment entry includes pods owned via its
106    /// ReplicaSets.
107    fn get_pods_by_controller(
108        pods: &[Arc<Pod>],
109        owner_ref_by_uid: &HashMap<Uid, OwnerReference>,
110    ) -> HashMap<Uid, Vec<Arc<Pod>>> {
111        let mut out: HashMap<Uid, Vec<Arc<Pod>>> = HashMap::new();
112        for pod in pods {
113            let Some(pod_uid) = pod.metadata.uid.as_deref() else {
114                continue;
115            };
116            let chain = Self::walk_up_in(owner_ref_by_uid, pod_uid);
117            for parent in chain {
118                out.entry(parent.uid.as_str().into())
119                    .or_default()
120                    .push(pod.clone());
121            }
122        }
123        out
124    }
125
126    /// Get all pods that are controlled by the controller at the given
127    /// `controller_uid`.
128    pub fn pods_by_controller(&self, controller_uid: &str) -> &[Arc<Pod>] {
129        self.pods_by_controller
130            .get(controller_uid)
131            .map_or(&[], Vec::as_slice)
132    }
133
134    /// Get all Jobs directly controlled by the controller at the given UID.
135    pub fn jobs_by_controller(&self, controller_uid: &str) -> &[Arc<Job>] {
136        self.jobs_by_controller
137            .get(controller_uid)
138            .map_or(&[], Vec::as_slice)
139    }
140
141    fn get_jobs_by_controller(
142        jobs: &[Arc<Job>],
143        owner_ref_by_uid: &HashMap<Uid, OwnerReference>,
144    ) -> HashMap<Uid, Vec<Arc<Job>>> {
145        let mut out: HashMap<Uid, Vec<Arc<Job>>> = HashMap::new();
146        for job in jobs {
147            let Some(uid) = job.metadata.uid.as_deref() else {
148                continue;
149            };
150            let Some(controller) = owner_ref_by_uid.get(uid) else {
151                continue;
152            };
153            out.entry(controller.uid.as_str().into())
154                .or_default()
155                .push(job.clone());
156        }
157        out
158    }
159
160    /// Like [`Self::walk_up()`] but operates on a borrowed map directly so that
161    /// it can be called during construction before the graph exists.
162    fn walk_up_in<'a>(
163        owner_ref_by_uid: &'a HashMap<Uid, OwnerReference>,
164        start: &str,
165    ) -> Vec<&'a OwnerReference> {
166        let mut chain = Vec::new();
167        let mut seen = HashSet::new();
168        let mut cur = owner_ref_by_uid.get(start);
169        while let Some(owner_ref) = cur {
170            if !seen.insert(owner_ref.uid.as_str()) {
171                break;
172            }
173            chain.push(owner_ref);
174            cur = owner_ref_by_uid.get(owner_ref.uid.as_str());
175        }
176        chain
177    }
178
179    /// Walk the ownership chain from `start`, returning the
180    /// [`OwnerReference`]s from the direct controller up to the root.
181    ///
182    /// Looks up the controller of `start`, then that controller's controller
183    /// and so on, until it reaches an object with no controlling owner (and
184    /// thus not in `owner_ref_by_uid`).
185    ///
186    /// The result is ordered nearest-first (so: a Pod owned by a ReplicaSet
187    /// owned by a Deployment will yield a vector containing first the
188    /// ReplicaSet and then the Deployment).
189    ///
190    /// The chain may be incomplete: if an owner hasn't been observed by the
191    /// watch yet, the walk simply stops there (eventual consistency, not an
192    /// error). Kubernetes should never produce a cycle, but this is
193    /// nevertheless guarded against; the walk always terminates.
194    pub fn walk_up(&self, start: &str) -> Vec<&OwnerReference> {
195        Self::walk_up_in(&self.owner_ref_by_uid, start)
196    }
197}

205    #[test]
206    fn walk_up() {
207        let graph = owner_graph(&[
208            ("pod-uid", owner_ref("ReplicaSet", "rs", "rs-uid")),
209            ("rs-uid", owner_ref("Deployment", "deploy", "deploy-uid")),
210        ]);
211
212        // Nearest first: ReplicaSet then Deployment
213        let chain = graph.walk_up("pod-uid");
214        assert_eq!(chain.len(), 2);
215        assert_eq!(chain[0].uid, "rs-uid");
216        assert_eq!(chain[1].uid, "deploy-uid");
217
218        // Starting mid chain
219        let chain = graph.walk_up("rs-uid");
220        assert_eq!(chain.len(), 1);
221        assert_eq!(chain[0].uid, "deploy-uid");
222
223        // Unknown start should not panic
224        assert!(graph.walk_up("definitely-does-not-exist").is_empty());
225    }

227    #[test]
228    fn walk_up_no_infinite_cycles() {
229        let graph = owner_graph(&[
230            ("a", owner_ref("SomeCrd", "b", "b")),
231            ("b", owner_ref("SomeCrd", "a", "a")),
232        ]);
233        let chain = graph.walk_up("a");
234        // We start at a, go to b, see a again and stop.
235        assert_eq!(chain.len(), 2);
236        assert_eq!(chain[0].uid, "b");
237        assert_eq!(chain[1].uid, "a");
238    }

240    /// Objects get mapped by their uid correctly to their owner reference and
241    /// are dropped from the map if they have no owner reference that is marked
242    /// as a controller.
243    #[test]
244    fn map_object_uids_to_owner_ref() {
245        let pod1 = pod_owned_by("pod1", "pod1-uid", owner_ref("ReplicaSet", "rs", "rs-uid"));
246        let rs = replicaset_owned_by("rs", "rs-uid", owner_ref("Deployment", "dep", "dep-uid"));
247        let job = job_owned_by("job1", "job1-uid", owner_ref("CronJob", "cj", "cj-uid"));
248
249        // Non-controller owner references are skipped
250        let mut non_controller = owner_ref("ReplicaSet", "rs", "rs-uid");
251        non_controller.controller = Some(false);
252        let pod_owned_by_non_controller = pod_owned_by("pod-2", "pod2-uid", non_controller);
253
254        let map = OwnerGraph::map_object_uids_to_owner_ref(
255            &[pod1.into(), pod_owned_by_non_controller.into()],
256            &[rs.into()],
257            &[job.clone().into()],
258        );
259
260        assert_eq!(map.len(), 3); // the pod with no controller is dropped
261        assert_eq!(map["pod1-uid"].uid, "rs-uid");
262        assert_eq!(map["rs-uid"].uid, "dep-uid");
263        assert_eq!(map["job1-uid"].uid, "cj-uid");
264
265        let jobs = [job.into()];
266        let jobs_by_controller = OwnerGraph::get_jobs_by_controller(&jobs, &map);
267        assert_eq!(jobs_by_controller["cj-uid"].len(), 1);
268    }

272        let pods = [
273            pod_owned_by("pod1", "pod1-uid", owner_ref("ReplicaSet", "rs", "rs-uid")).into(),
274            pod_owned_by("pod2", "pod2-uid", owner_ref("ReplicaSet", "rs", "rs-uid")).into(),
275        ];
276        let replicasets =
277            [
278                replicaset_owned_by("rs", "rs-uid", owner_ref("Deployment", "dep", "dep-uid"))
279                    .into(),
280            ];
281        let jobs = [job_owned_by("job1", "job1-uid", owner_ref("CronJob", "cj", "cj-uid")).into()];
282        let owner_ref_by_uid = OwnerGraph::map_object_uids_to_owner_ref(&pods, &replicasets, &jobs);
283        let pods_owned_by_rs = OwnerGraph::get_pods_by_controller(&pods, &owner_ref_by_uid);
284
285        assert_eq!(pods_owned_by_rs.len(), 2); // two controllers: rs-uid and dep-uid
286        assert_eq!(pods_owned_by_rs["dep-uid"].len(), 2); // both pods show up
287        assert_eq!(pods_owned_by_rs["rs-uid"].len(), 2); // both pods show up
288    }
